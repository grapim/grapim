import pg from 'pg';

const pool = new pg.Pool({ connectionString: process.env.DATABASE_URL });

export const db = {
  async one(sql: string, params: unknown[] = []) {
    const { rows } = await pool.query(sql, params);
    return rows[0] ?? null;
  },
  async many(sql: string, params: unknown[] = []) {
    const { rows } = await pool.query(sql, params);
    return rows;
  },
};
